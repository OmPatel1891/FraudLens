# Serving-only dependencies.
#
# requirements.txt is the full development stack: MLflow, Jupyter, Streamlit,
# Evidently, plotting and the Kaggle client. None of that is reachable from
# api/main.py, and installing it into the serving image cost roughly 2.5 GB.
# The Dockerfile trains in a builder stage using the full file, then copies only
# the artifacts into a runtime stage built from this one.
#
# xgboost is here even though the current best model is LightGBM: model
# selection is data-dependent, and an image that cannot unpickle the model it
# was shipped with is a bad trade for a few hundred megabytes.

# Compatible-release pins on everything whose objects live inside the pickled
# artifacts. Estimators unpickle by class reference, so a minor-version bump
# rebuilds them from different code and can change predictions without raising.
# These must stay in step with requirements.txt; fraudlens/versions.py records
# what trained the model and the API logs any mismatch at load.
numpy~=2.4.0
scipy~=1.16.3
scikit-learn~=1.8.0
xgboost~=3.4.0
lightgbm~=4.7.0
joblib~=1.5.3

pandas>=2.1,<3
shap>=0.45,<1

fastapi>=0.111,<1
uvicorn[standard]>=0.29,<1
python-multipart>=0.0.9
pydantic>=2.7,<3
